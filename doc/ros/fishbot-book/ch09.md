#### (1)上位机 micro-ROS-Agent + micro_ros_msgs
```sh
# 下载
chapt09$ mkdir -p fishbot_ws/src

chapt09$ cd fishbot_ws/src
chapt09/fishbot_ws/src$ git clone https://github.com/micro-ROS/micro-ROS-Agent.git -b $ROS_DISTRO 
chapt09/fishbot_ws/src$ git clone https://github.com/micro-ROS/micro_ros_msgs.git  -b $ROS_DISTRO

# 构建运行
chapt09/fishbot_ws/src$ cd ..
chapt09/fishbot_ws$ colcon build
chapt09/fishbot_ws$ source install/setup.bash
# 
chapt09/fishbot_ws$ ros2 run micro_ros_agent micro_ros_agent udp4 --port 8888
```
